% Kindle-friendly small-format preamble.
% Page geometry matches a 6" e-reader: narrow margins, no gutter, ragged
% bottom. A PDF has no real reflow, so the page itself must already be
% Kindle-shaped and the type sized for it.
%
% Base size is 6pt on a 90mm page. That sounds tiny and isn't: what the eye
% judges is the ratio of type to measure, and 6pt across an 80mm text block
% is the same ratio as 12pt across 160mm -- an ordinary trade paperback.
% The earlier 12pt setting gave a ~30-character measure, which is newspaper-
% column narrow and reads badly: constant hyphenation, loose interword space,
% and displayed equations wider than the column.
%
% The `fontsize` package is used rather than \fontsize{6}{...}\selectfont
% because it regenerates the whole size hierarchy -- \small, \footnotesize,
% section headings, and crucially the math script and scriptscript sizes.
% Setting the body size by hand leaves subscripts sized for a 12pt document.
\documentclass[oneside]{book}
\usepackage[fontsize=6pt]{fontsize}

\usepackage[
  paperwidth=90mm, paperheight=120mm,
  top=4mm, bottom=5mm, left=4mm, right=4mm,
  headsep=2mm, footskip=5mm
]{geometry}

\usepackage{amsmath,amssymb,amsthm,amsfonts}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage{microtype}
\usepackage[hidelinks]{hyperref}
\usepackage{fancyhdr}
\usepackage{titlesec}
\linespread{1.15}   % a little air; e-ink contrast is lower than paper

% Display skips stay tight, but less desperately so than at 12pt.
\setlength{\parindent}{1em}
\setlength{\abovedisplayskip}{3pt plus 1pt minus 1pt}
\setlength{\belowdisplayskip}{3pt plus 1pt minus 1pt}
\setlength{\abovedisplayshortskip}{1.5pt}
\setlength{\belowdisplayshortskip}{1.5pt}
\raggedbottom
\sloppy   % safety net; at 6pt the measure is wide enough to justify properly

\titleformat{\chapter}[block]{\bfseries\large}{\thechapter.}{0.5em}{}
\titlespacing*{\chapter}{0pt}{0pt}{1em}
\titleformat{\section}[block]{\bfseries\normalsize}{\thesection}{0.5em}{}
\titlespacing*{\section}{0pt}{1em}{0.5em}

\pagestyle{fancy}
\fancyhf{}
\fancyfoot[C]{\footnotesize\thepage}
\renewcommand{\headrulewidth}{0pt}

\newtheorem{theorem}{Theorem}[chapter]
\newtheorem{lemma}[theorem]{Lemma}
\newtheorem{definition}[theorem]{Definition}
\newtheorem{corollary}[theorem]{Corollary}

% Every unresolved OCR gap is rendered VISIBLY. This is deliberate: a silent
% gap is indistinguishable from correct text, and this book's whole risk is
% invented mathematics. Nothing ships until the count of these is zero.
\newcommand{\UNSURE}[1]{\textbf{[?\,\emph{#1}]}\linebreak[0]}
