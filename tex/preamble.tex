% Kindle-friendly small-format preamble.
% Page geometry matches a 6" e-reader: narrow margins, generous leading,
% no gutter, ragged bottom. Reflow on a Kindle comes from the small trim size
% plus a large body font, not from the PDF being reflowable text (PDF has no
% real reflow) -- so the page itself must already be Kindle-shaped.
\documentclass[12pt,oneside]{book}

\usepackage[
  paperwidth=90mm, paperheight=120mm,
  top=5mm, bottom=6mm, left=5mm, right=5mm,
  headsep=2mm, footskip=5mm
]{geometry}

\usepackage{amsmath,amssymb,amsthm,amsfonts}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage{microtype}
\usepackage[hidelinks]{hyperref}
\usepackage{fancyhdr}
\usepackage{titlesec}

% Small pages need small display skips or every equation blows a page break.
\setlength{\parindent}{1em}
\setlength{\abovedisplayskip}{4pt}
\setlength{\belowdisplayskip}{4pt}
\setlength{\abovedisplayshortskip}{2pt}
\setlength{\belowdisplayshortskip}{2pt}
\raggedbottom
\sloppy   % 90mm measure: prefer loose lines over overfull boxes

\titleformat{\chapter}[block]{\bfseries\large}{\thechapter.}{0.5em}{}
\titlespacing*{\chapter}{0pt}{0pt}{1em}
\titleformat{\section}[block]{\bfseries\normalsize}{\thesection}{0.5em}{}
\titlespacing*{\section}{0pt}{1em}{0.5em}

\pagestyle{fancy}
\fancyhf{}
\fancyfoot[C]{\footnotesize\thepage}
\renewcommand{\headrulewidth}{0pt}

\newtheorem{theorem}{Theorem}[chapter]
\newtheorem{lemma}[theorem]{Lemma}
\newtheorem{definition}[theorem]{Definition}
\newtheorem{corollary}[theorem]{Corollary}

% Every unresolved OCR gap is rendered VISIBLY. This is deliberate: a silent
% gap is indistinguishable from correct text, and this book's whole risk is
% invented mathematics. Nothing ships until the count of these is zero.
\newcommand{\UNSURE}[1]{\textbf{[?~#1]}}
